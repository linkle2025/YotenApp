        validated_data.pop('password2')
        user = User.objects.create_user(
            username=validated_data['username'],
            email=validated_data['email'],
            password=validated_data['password']
        )
        return user

"""
四柱推命計算ロジック
生年月日時から四柱（年柱・月柱・日柱・時柱）を計算
"""

# 十干（天干）
HEAVENLY_STEMS = ['甲', '乙', '丙', '丁', '戊', '己', '庚', '辛', '壬', '癸']

# 十二支（地支）
EARTHLY_BRANCHES = ['子', '丑', '寅', '卯', '辰', '巳', '午', '未', '申', '酉', '戌', '亥']

# 月の節月（立春から始まる）
SOLAR_TERMS = [
    (2, 4),   # 立春（2月4日頃）
    (3, 6),   # 啓蟄（3月6日頃）
    (4, 5),   # 清明（4月5日頃）
    (5, 6),   # 立夏（5月6日頃）
    (6, 6),   # 芒種（6月6日頃）
    (7, 7),   # 小暑（7月7日頃）
    (8, 8),   # 立秋（8月8日頃）
    (9, 8),   # 白露（9月8日頃）
    (10, 8),  # 寒露（10月8日頃）
    (11, 7),  # 立冬（11月7日頃）
    (12, 7),  # 大雪（12月7日頃）
    (1, 6),   # 小寒（1月6日頃）
]


def get_solar_month(year, month, day):
    """
    節月を取得（立春から始まる）
    簡易版：月の前半は前月、後半は当月とする
    """
    # 立春（2月4日頃）を基準に、2月4日以降を1月とする
    if month == 1:
        if day < 6:  # 小寒より前
            return 12  # 前年の12月
        else:
            return 1
    elif month == 2:
        if day < 4:  # 立春より前
            return 1
        else:
            return 2
    else:
        # 3月以降は、月の前半（15日まで）は前月、後半（16日以降）は当月
        if day < 15:
            return month - 1
        else:
            return month


def calculate_year_pillar(year):
    """
    年柱を計算
    年柱は生年の十干十二支
    """
    # 基準年（1984年）を甲子年とする
    base_year = 1984
    offset = (year - base_year) % 60
    
    heavenly_stem_index = offset % 10
    earthly_branch_index = offset % 12
    
    return {
        'heavenly_stem': HEAVENLY_STEMS[heavenly_stem_index],
        'earthly_branch': EARTHLY_BRANCHES[earthly_branch_index]
    }


def calculate_month_pillar(year, month, day):
    """
    月柱を計算
    月柱は節月の十干十二支
    """
    solar_month = get_solar_month(year, month, day)
    
    # 年柱の天干から月柱の天干を計算
    year_pillar = calculate_year_pillar(year)
    year_stem_index = HEAVENLY_STEMS.index(year_pillar['heavenly_stem'])
    
    # 月柱の天干は年柱の天干から決まる（簡易版）
    # 実際の計算はより複雑だが、ここでは簡易版を使用
    month_stem_index = (year_stem_index * 2 + solar_month - 1) % 10
    
    # 月柱の地支は節月から決まる（立春=寅月=3）
    month_branch_index = (solar_month + 1) % 12
    
    return {
        'heavenly_stem': HEAVENLY_STEMS[month_stem_index],
        'earthly_branch': EARTHLY_BRANCHES[month_branch_index]
    }


def calculate_day_pillar(year, month, day):
    """
    日柱を計算
    日柱は生年月日から計算（簡易版：実際はより複雑な計算が必要）
    """
    from datetime import datetime
    
    # 基準日（1900年1月1日）を甲子日とする
    base_date = datetime(1900, 1, 1)
    target_date = datetime(year, month, day)
    days_diff = (target_date - base_date).days
    
    # 60日周期で循環
    offset = days_diff % 60
    
    heavenly_stem_index = offset % 10
    earthly_branch_index = offset % 12
    
    return {
        'heavenly_stem': HEAVENLY_STEMS[heavenly_stem_index],
        'earthly_branch': EARTHLY_BRANCHES[earthly_branch_index]
    }


def calculate_hour_pillar(year, month, day, hour):
    """
    時柱を計算
    時柱は生時と日柱の天干から計算
    """
    day_pillar = calculate_day_pillar(year, month, day)
    day_stem_index = HEAVENLY_STEMS.index(day_pillar['heavenly_stem'])
    
    # 時柱の地支は生時から決まる（23-1時=子、1-3時=丑、...）
    hour_branch_index = ((hour + 1) // 2) % 12
    
    # 時柱の天干は日柱の天干から決まる（簡易版）
    hour_stem_index = (day_stem_index * 2 + hour_branch_index) % 10
    
    return {
        'heavenly_stem': HEAVENLY_STEMS[hour_stem_index],
        'earthly_branch': EARTHLY_BRANCHES[hour_branch_index]
    }


def calculate_fortune(birth_year, birth_month, birth_day, birth_hour):
    """
    四柱推命を計算
    """
    year_pillar = calculate_year_pillar(birth_year)
    month_pillar = calculate_month_pillar(birth_year, birth_month, birth_day)
    day_pillar = calculate_day_pillar(birth_year, birth_month, birth_day)
    hour_pillar = calculate_hour_pillar(birth_year, birth_month, birth_day, birth_hour)
    
    return {
        'year_pillar': year_pillar,
        'month_pillar': month_pillar,
        'day_pillar': day_pillar,
        'hour_pillar': hour_pillar,
    }

        validated_data['user'] = self.context['request'].user
        
        return super().create(validated_data)

    }

  },

api.interceptors.response.use(

        const refreshToken = localStorage.getItem('refresh_token');
        if (refreshToken) {

            refresh: refreshToken,
          });

          localStorage.setItem('access_token', access);

          return api(originalRequest);
        }
      } catch (refreshError) {

        localStorage.removeItem('refresh_token');

      }
    }

  }

const AuthContext = createContext();

export const useAuth = () => {
  const context = useContext(AuthContext);
  if (!context) {
    throw new Error('useAuth must be used within an AuthProvider');
  }
  return context;
};

export const AuthProvider = ({ children }) => {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const loadUser = async () => {
      if (authService.isAuthenticated()) {
        try {
          const userData = await authService.getCurrentUser();
          setUser(userData);
        } catch (error) {
          authService.logout();
        }
      }
      setLoading(false);
    };
    loadUser();
  }, []);

  const login = async (username, password) => {
    await authService.login(username, password);
    const userData = await authService.getCurrentUser();
    setUser(userData);
  };

  const register = async (username, email, password, password2) => {
    await authService.register(username, email, password, password2);
    await login(username, password);
  };

  const logout = () => {
    authService.logout();
    setUser(null);
  };

  return (
    <AuthContext.Provider value={{ user, login, register, logout, loading }}>
      {children}
    </AuthContext.Provider>
  );
};

      const data = await fortuneService.getDetail(id);
      setCalculation(data);

      setError('詳細の取得に失敗しました。');
    } finally {
      setLoading(false);
    }
  };

  const handleToggleFavorite = async () => {
    try {
      const result = await fortuneService.toggleFavorite(id);
      setCalculation({ ...calculation, is_favorite: result.is_favorite });
    } catch (error) {
      alert('お気に入りの更新に失敗しました。');
    }

  const handleShare = async () => {

      alert('共有URLの生成に失敗しました。');
